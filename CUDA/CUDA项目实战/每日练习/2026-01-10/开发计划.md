```
from torch.utils.cpp_extension import load
lib = load(
    name = "gemm",
    sources=["gemm.cu"],
    extra_cuda_cflags = [
        "-O3",
        "-gencode", "arch=compute_75,code=sm_75"
    ],
    extra_cflags = [
        "-std=C++17"
    ],
    verbose = True
)
```
